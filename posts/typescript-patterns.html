TypeScript: 9 أنماط تحوّل أنواعك من عبء إلى توثيق حيّ

الأنواع الجيدة تمنع الأخطاء قبل التشغيل وتشرح الكود بلا تعليقات. أنماط عملية: الأصناف المميّزة، الأنواع الشرطية، Generics، وأنواع النتائج الصريحة.

مفريق MyTechol
· 2025-11-14 · 12 دقيقة قراءة · متوسط
TS

معظم مشاريع TypeScript تكتب أنواعاً تُرضي المُصرِّف فقط: any هنا، as هناك، وstring لكل شيء. النتيجة: لا حماية حقيقية، ولا توثيق. الأنواع الجيدة تُصمَّم، لا تُضاف.

لماذا الأنواع الجيدة نادرة؟

لأن كتابتها تأخذ وقتاً في البداية وتوفّر وقتاً لاحقاً — والعقل البشري سيئ في تقدير المنافع المؤجّلة. لكن كل نوع دقيق يمنع فئة كاملة من الأخطاء إلى الأبد.

السؤال الذي يوجّه كل قرار نوعي: «هل يمكن تمثيل حالة غير صالحة بهذا النوع؟» إن كانت الإجابة نعم، فالنوع ناقص.

1. الأصناف المميّزة: استبدل booleans متعددة

{{
discriminated.ts
// ✗ يمكن تمثيل حالات مستحيلة
type State = {{ loading: boolean; data?: User; error?: string }};
// loading && error معاً؟ مسموح. لا معنى له.

// ✓ كل حالة ممثّلة بدقة، والمستحيل غير قابل للتمثيل
type State =
  | {{ status: "idle" }}
  | {{ status: "loading" }}
  | {{ status: "success"; data: User }}
  | {{ status: "error"; message: string }};

function render(s: State) {{
  switch (s.status) {{
    case "idle":    return "ابدأ البحث";
    case "loading": return "جارٍ التحميل…";
    case "success": return s.data.name;     // data مضمون هنا
    case "error":   return s.message;       // message مضمون هنا
  }}
}}
}}

الفائدة المزدوجة: المُصرِّف يمنع الحالات المستحيلة، وعبارة switch تضمن أنك عالجت كل حالة — إن أضفت حالة جديدة، يظهر خطأ في كل مكان لم يُحدَّث.

2. نمط Result: الأخطاء جزء من النوع

{{
result.ts
type Ok<T>  = {{ ok: true;  value: T }};
type Err<E> = {{ ok: false; error: E }};
type Result<T, E = Error> = Ok<T> | Err<E>;

async function fetchUser(id: string): Promise<Result<User, "not_found" | "network">> {{
  try {{
    const res = await fetch(`/api/users/${{id}}`);
    if (res.status === 404) return {{ ok: false, error: "not_found" }};
    return {{ ok: true, value: await res.json() }};
  }} catch {{
    return {{ ok: false, error: "network" }};
  }}
}}

// المستدعي مجبر على معالجة الحالتين
const r = await fetchUser("42");
if (!r.ok) return showError(r.error);   // r.error مقيّد قيمياً
console.log(r.value.email);
}}
هذا النمط شائع في Rust وGo، ويحلّ مشكلة try/catch التي ينساها المطور بسهولة. الخطأ الذي لا يمكن تجاهله هو الخطأ الذي لن يصل للمستخدم.

3. الأنواع الموسومة: امنع تمرير معرّف مكان آخر

{{
branded.ts
// المشكلة: كل المعرّفات من نوع string — يُمكن تمرير userId في مكان orderId
type UserId  = string & {{ readonly __brand: "UserId" }};
type OrderId = string & {{ readonly __brand: "OrderId" }};

const asUserId  = (s: string) => s as UserId;
const asOrderId = (s: string) => s as OrderId;

function getUser(id: UserId) {{ /* ... */ }}

const uid = asUserId("u_123");
const oid = asOrderId("o_456");

getUser(uid);   // ✓
// getUser(oid); // ✗ خطأ تضمين: OrderId ليس UserId
}}

مفيد تحديداً في الأنظمة التي تتعامل مع أرقام وحسابات ومعرّفات كثيرة متشابهة الشكل.

4. Generics: قيود تمنع الفوضى

{{
generics.ts
// ✗ generic بلا قيد: T يمكن أن يكون أي شيء، فلا تستطيع الوصول لأي خاصية
function first<T>(arr: T[]): T | undefined {{ return arr[0]; }}

// ✓ قيد يوثّق المتطلبات ويسمح بالوصول الآمن
function groupBy<T, K extends string | number>(
  items: T[],
  key: (item: T) => K
): Record<K, T[]> {{
  const out = {{}} as Record<K, T[]>;
  for (const item of items) {{
    const k = key(item);
    (out[k] ??= []).push(item);
  }}
  return out;
}}

const byCity = groupBy(users, u => u.city);
// byCity["القاهرة"] من نوع User[] — بلا any ولا as
}}

5. الأنواع المركّبة: اشتقّ لا تكرّر

{{
mapped.ts
interface User {{
  id: number;
  email: string;
  name: string;
  role: "admin" | "editor" | "viewer";
  createdAt: Date;
}}

// اشتق الحاجات من مصدر واحد — بلا تكرار
type CreateUserInput = Omit<User, "id" | "createdAt">;
type UserDto        = Omit<User, "createdAt"> & {{ createdAt: string }};
type UserPatch      = Partial<Pick<User, "name" | "role">>;
type AdminOnly      = Required<Pick<User, "email">> & {{ role: "admin" }};
}}

6. اشتقاق الأصناف من ثوابت

{{
const.ts
const ROLES = ["admin", "editor", "viewer"] as const;
type Role = typeof ROLES[number];              // "admin" | "editor" | "viewer"

function isRole(v: string): v is Role {{
  return (ROLES as readonly string[]).includes(v);
}}

// الآن القائمة والقيمة النوعية مصدرهما واحد — لا يمكن أن يتباعدا
}}

7. نفي الأنواع (Type Guards)

{{
guards.ts
function assertDefined<T>(v: T | null | undefined, msg: string): asserts v is T {{
  if (v == null) throw new Error(msg);
}}

const el = document.querySelector<HTMLInputElement>("#email");
assertDefined(el, "عنصر #email غير موجود");
el.value;   // آمن تماماً، بلا ! مزعجة
}}

8. دوال مساعدة صغيرة عالية القيمة

{{
utils.ts
// MakeCertain: يزيل undefined من كائن كامل
type MakeCertain<T> = {{ [K in keyof T]-?: NonNullable<T[K]> }};

// NonEmpty: يمنع تمرير مصفوفة فارغة
type NonEmpty<T> = readonly [T, ...T[]];

function first<T>(arr: NonEmpty<T>): T {{ return arr[0]; }}
}}

9. اضبط الإعدادات الصارمة

{{
tsconfig.json
{{
  "compilerOptions": {{
    "strict": true,
    "noUncheckedIndexedAccess": true,
    "noImplicitOverride": true,
    "exactOptionalPropertyTypes": true,
    "verbatimModuleSyntax": true
  }}
}}
}}

noUncheckedIndexedAccess وحده يكشف أخطاء حقيقية في كل مشروع تقريباً — لأن arr[0] قد يكون undefined فعلياً.

الخطوة التالية: افتح مشروعك، وشغّل grep -rn ": any" src/. استبدل واحداً منها اليوم بنوع حقيقي. عادة واحدة صغيرة تتكرّر تُغيّر جودة قاعدة الكود بالكامل.
