SQL للطلاب: من SELECT إلى Window Functions بمثال واحد متصل

مسار متصل بمتجر إلكتروني واحد: تجميع، ربط، مجموعات فرعية، ثم دوال النوافذ التي تفصل المبتدئ عن المحترف — مع تمارين بعد كل خطوة.

مفريق MyTechol

SQL

SQL ليست لغة «قديمة يجب تعلّمها». هي اللغة التي تُخاطب بها أكبر كمية بيانات في العالم، وهي أسهل بكثير مما يُظن إذا تعلّمتها بترتيب صحيح — لا كقائمة أوامر.

سنستخدم متجراً إلكترونياً صغيراً من 3 جداول، وننتقل من أبسط استعلام إلى أعقد تحليل.

الجدول الذي سنعمل عليه

{{
schema.sql
CREATE TABLE customers (
  id          serial PRIMARY KEY,
  name        text NOT NULL,
  city        text,
  created_at  date NOT NULL DEFAULT CURRENT_DATE
);

CREATE TABLE products (
  id        serial PRIMARY KEY,
  name      text NOT NULL,
  price     numeric(10,2) NOT NULL,
  category  text
);

CREATE TABLE orders (
  id           serial PRIMARY KEY,
  customer_id  int REFERENCES customers(id),
  product_id   int REFERENCES products(id),
  qty          int NOT NULL CHECK (qty > 0),
  amount       numeric(10,2) NOT NULL,
  ordered_at   timestamp NOT NULL DEFAULT now()
);
}}

SELECT و WHERE: الأساس الصحيح

ترتيب تنفيذ SQL يختلف عن ترتيب كتابتها — وهذا يفسّر كثيراً من الأخطاء:

ترتيب الكتابةSELECT → FROM → WHERE → GROUP BY → HAVING → ORDER BY
ترتيب التنفيذFROM → WHERE → GROUP BY → HAVING → SELECT → ORDER BY
لهذا لا يمكنك استخدام اسم مستعار (alias) من SELECT داخل WHERE: القاعدة لم تحسبه بعد في تلك المرحلة.
{{
select.sql
SELECT name, city, price
FROM products
WHERE category = 'لابتوب' AND price BETWEEN 2000 AND 5000
ORDER BY price DESC
LIMIT 5;
}}

التجميع: الأسئلة الحقيقية

{{
group.sql
-- كم عميل في كل مدينة؟
SELECT city, COUNT(*) AS customers
FROM customers
GROUP BY city
ORDER BY customers DESC;

-- إجمالي المبيعات حسب الفئة (HAVING بعد التجميع)
SELECT p.category,
       COUNT(*)                        AS orders,
       SUM(o.amount)                   AS revenue,
       ROUND(AVG(o.amount), 2)         AS avg_order
FROM orders o
JOIN products p ON p.id = o.product_id
GROUP BY p.category
HAVING SUM(o.amount) > 10000
ORDER BY revenue DESC;
}}

الفرق الجوهري: WHERE يفلتر الصفوف قبل التجميع، وHAVING يفلتر المجموعات بعده.

الربط: أنواعه ومتى تستخدم كل واحد

النوعماذا يُرجع
INNER JOINالصفوف المتطابقة في الجدولين فقط
LEFT JOINكل صفوف اليسار + المطابق من اليمين (أو NULL)
FULL OUTER JOINكل شيء من الجهتين
CROSS JOINكل التوليفات الممكنة
{{
left-join.sql
-- كل العملاء، حتى من لم يشترِ قط (LEFT JOIN)
SELECT c.name, c.city,
       COUNT(o.id)               AS order_count,
       COALESCE(SUM(o.amount), 0) AS spent
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.id
GROUP BY c.id, c.name, c.city
ORDER BY spent DESC;
}}
خطأ شائع: وضع شرط على الجدول الأيمن داخل WHERE يحوّل LEFT JOIN إلى INNER JOIN ويكسر النتيجة. ضع الشرط في ON بدلاً من ذلك.

دوال النوافذ: الخط الفاصل

التجميع يدمج الصفوف ويفقدها. دالة النافذة تحتفظ بكل صف وتضيف إليه معلومة محسوبة ضمن مجموعة.

{{
window.sql
-- لكل عميل: ترتيبه ضمن مدينته حسب الإنفاق
SELECT c.name, c.city, SUM(o.amount) AS spent,
       RANK() OVER (PARTITION BY c.city ORDER BY SUM(o.amount) DESC) AS rank_in_city,
       SUM(o.amount) OVER (PARTITION BY c.city) AS city_total,
       ROUND(100.0 * SUM(o.amount) / SUM(SUM(o.amount)) OVER (PARTITION BY c.city), 1) AS pct
FROM customers c
JOIN orders o ON o.customer_id = c.id
GROUP BY c.id, c.name, c.city
ORDER BY c.city, rank_in_city;
}}

الدوال الأكثر فائدة:

  • ROW_NUMBER() — ترقيم متسلسل بلا تكرار.
  • RANK() — ترتيب مع فجوات عند التعادل.
  • LAG() / LEAD() — الصف السابق واللاحق (مثالي للنموّ).
  • SUM() OVER (...) — إجمالي تراكمي.
{{
growth.sql
-- نموّ الإيرادات شهراً بشهر
SELECT date_trunc('month', ordered_at) AS month,
       SUM(amount) AS revenue,
       LAG(SUM(amount)) OVER (ORDER BY date_trunc('month', ordered_at)) AS prev_month,
       ROUND(100.0 * (SUM(amount) - LAG(SUM(amount)) OVER (ORDER BY date_trunc('month', ordered_at)))
             / NULLIF(LAG(SUM(amount)) OVER (ORDER BY date_trunc('month', ordered_at)), 0), 1) AS growth_pct
FROM orders
GROUP BY 1
ORDER BY 1;
}}

CTEs: اجعل الاستعلام يُقرأ كالجملة

{{
cte.sql
WITH monthly AS (
  SELECT date_trunc('month', ordered_at) AS m, SUM(amount) AS rev
  FROM orders GROUP BY 1
),
ranked AS (
  SELECT m, rev, LAG(rev) OVER (ORDER BY m) AS prev FROM monthly
)
SELECT to_char(m, 'YYYY-MM') AS month,
       rev,
       CASE
         WHEN prev IS NULL THEN 'أول شهر'
         WHEN rev > prev   THEN '▲ نمو'
         ELSE '▼ تراجع'
       END AS trend
FROM ranked ORDER BY m;
}}

تمارين (حلّها بنفسك)

  1. اعرض أفضل 3 منتجات مبيعاً في كل فئة باستخدام ROW_NUMBER.
  2. احسب نسبة العملاء الذين اشتروا أكثر من مرة.
  3. اعرض متوسط الفترة بين طلبين متتاليين لكل عميل (LAG).
  4. اعرض العملاء الذين لم يشتروا شيئاً في آخر 90 يوماً.
الخطوة التالية: أنشئ هذه الجداول في SQLite أو PostgreSQL محلياً (5 دقائق)، وأدخل 20 صفاً يدوياً، ونفّذ التمرين الأول. التجربة الفعلية تُغني عن قراءة عشرة دروس.
