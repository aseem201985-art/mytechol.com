بناء API احترافي بـ Node.js و Express خطوة بخطوة

سنبني واجهة برمجية كاملة للمهام: توجيه، تحقق من المدخلات، معالجة أخطاء موحّدة، ومصادقة بـ JWT — ببنية مجلدات تتحمل النمو.

أفريق MyTechol
· 2025-11-22 · 16 دقيقة قراءة · متوسط
API

معظم دروس Express تتوقف عند «مرحباً بالعالم». هذا الدرس يبدأ من حيث تتوقف تلك الدروس: كيف تكتب API يمكن نشره فعلاً — مع أخطاء مفهومة، مدخلات موثوقة، ورمز ينمو دون أن ينهار.

1. التهيئة

mkdir task-api && cd task-api
npm init -y
npm i express zod jsonwebtoken bcryptjs dotenv
npm i -D nodemon

# package.json → أضف
"scripts": {
  "dev": "nodemon src/server.js",
  "start": "node src/server.js"
}

استخدامات الحزم واضحة من أسمائها: express للخادم، zod للتحقق من المدخلات، jsonwebtoken وbcryptjs للمصادقة، dotenv للمتغيرات السرية.

2. بنية المجلدات — القرار الأهم

يمكنك كتابة كل شيء في ملف واحد، وسيصبح غير قابل للصيانة بعد أسبوع. افصل من البداية:

src/
├── server.js          نقطة التشغيل
├── app.js             إعداد Express
├── routes/            تعريف المسارات
│   ├── auth.routes.js
│   └── tasks.routes.js
├── controllers/       منطق كل مسار
│   ├── auth.controller.js
│   └── tasks.controller.js
├── middleware/        طبقات مشتركة
│   ├── auth.js
│   └── error.js
└── utils/
    └── ApiError.js

الفصل ليس تنظيمياً فقط: كل طبقة تعرف مهمتها الواحدة. المسار يعرف العنوان، المتحكم يعرف المنطق، الوسيط يعرف ما يحدث قبل أو بعد الكل.

3. إعداد التطبيق

// src/app.js
import express from "express";
import authRoutes from "./routes/auth.routes.js";
import taskRoutes from "./routes/tasks.routes.js";
import { errorHandler } from "./middleware/error.js";

const app = express();

app.use(express.json({ limit: "1mb" }));

app.get("/health", (req, res) => res.json({ ok: true }));

app.use("/api/auth",  authRoutes);
app.use("/api/tasks", taskRoutes);

// يجب أن يكون آخر وسيط — بعده لا شيء
app.use(errorHandler);

export default app;
// src/server.js
import "dotenv/config";
import app from "./app.js";

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`API on :${PORT}`));

لاحظ مسار /health: نقطة فحص صغيرة يستخدمها أي نظام نشر ليعرف أن الخدمة حيّة. أضفها في كل مشروع من اليوم الأول.

4. معالجة أخطاء موحّدة

هذا الجزء يتجاهله معظم الدروس، وهو أكثر ما يفرّق بين API هاوٍ وآخر احترافي. أولاً، نوع خطأ خاص:

// src/utils/ApiError.js
export class ApiError extends Error {
  constructor(status, message, details = null) {
    super(message);
    this.status = status;
    this.details = details;
  }
}
// src/middleware/error.js
import { ApiError } from "../utils/ApiError.js";

export function errorHandler(err, req, res, next) {
  const status = err instanceof ApiError ? err.status : 500;

  // الأخطاء غير المتوقعة: سجّلها كاملة، ولا تكشف تفاصيلها للعميل
  if (status === 500) console.error("[unhandled]", err);

  res.status(status).json({
    error: {
      message: status === 500 ? "خطأ في الخادم" : err.message,
      details: err.details ?? undefined
    }
  });
}

النتيجة: كل استجابة خطأ في التطبيق لها الشكل نفسه. عميلك (أو واجهتك) يعرف دائماً أين يقرأ الرسالة.

قاعدة أمنية: لا تكشف رسالة الخطأ الحقيقية في حالة 500. قد تحتوي على مسار ملف أو استعلام قاعدة بيانات أو مفتاح. سجّلها في الخادم واعرض رسالة عامة للعميل.

5. التحقق من المدخلات بـ Zod

لا تثق بأي بيانات قادمة من الخارج. التحقق في أول سطر من المتحكم هو خط الدفاع:

// src/routes/tasks.routes.js
import { Router } from "express";
import { z } from "zod";
import { requireAuth } from "../middleware/auth.js";
import { createTask, listTasks } from "../controllers/tasks.controller.js";
import { ApiError } from "../utils/ApiError.js";

const router = Router();

const taskSchema = z.object({
  title: z.string().min(3, "العنوان قصير جداً").max(200),
  priority: z.enum(["low", "normal", "high"]).default("normal")
});

router.use(requireAuth);   // كل مسارات الملف تحتاج تسجيل دخول

router.get("/", listTasks);

router.post("/", (req, res, next) => {
  const result = taskSchema.safeParse(req.body);

  if (!result.success) {
    return next(new ApiError(400, "بيانات غير صحيحة", result.error.flatten()));
  }

  req.validated = result.data;   // البيانات الآن موثوقة
  next();
}, createTask);

export default router;

لاحظ req.validated: نضع البيانات المنقّاة في مكان منفصل، فلا يستخدم المتحكم req.body الخام أبداً. هذا يمنع تمرير حقول لم تتحقق منها عمداً أو سهواً.

6. المصادقة بـ JWT

// src/controllers/auth.controller.js
import bcrypt from "bcryptjs";
import jwt from "jsonwebtoken";
import { ApiError } from "../utils/ApiError.js";

// نفّذها بقاعدة بيانات حقيقية — مصفوفة للتوضيح فقط
const users = [];

export async function register(req, res, next) {
  const { email, password } = req.body;

  if (users.some(u => u.email === email)) {
    return next(new ApiError(409, "البريد مسجّل مسبقاً"));
  }

  const hash = await bcrypt.hash(password, 12);
  users.push({ id: Date.now(), email, password: hash });

  res.status(201).json({ message: "تم إنشاء الحساب" });
}

export async function login(req, res, next) {
  const { email, password } = req.body;
  const user = users.find(u => u.email === email);

  // نفس الرسالة لحالتَي «غير موجود» و«كلمة خاطئة»
  // لعدم تسريب أي بريد مسجّل
  if (!user || !(await bcrypt.compare(password, user.password))) {
    return next(new ApiError(401, "بيانات الدخول غير صحيحة"));
  }

  const token = jwt.sign(
    { sub: user.id },
    process.env.JWT_SECRET,
    { expiresIn: "7d" }
  );

  res.json({ token });
}
// src/middleware/auth.js
import jwt from "jsonwebtoken";
import { ApiError } from "../utils/ApiError.js";

export function requireAuth(req, res, next) {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : null;

  if (!token) return next(new ApiError(401, "مطلوب تسجيل الدخول"));

  try {
    req.user = jwt.verify(token, process.env.JWT_SECRET);
    next();
  } catch {
    next(new ApiError(401, "الرمز منتهي أو غير صالح"));
  }
}

ثلاث قواعد لا تتنازل عنها:

  • كلمة المرور تُشفّر دائماً بـ bcrypt، ولا تُخزَّن نصاً صريحاً أبداً.
  • JWT_SECRET في .env فقط، وليس في الكود ولا في Git.
  • نفس رسالة الخطأ لبريد غير مسجّل أو كلمة مرور خاطئة — وإلا صار API أداة لمعرفة من لديه حساب.

7. رموز الحالة: اتفق عليها وثبّتها

الرمزالمعنىمتى تستخدمه
200نجاحGET أو تحديث ناجح
201أُنشئبعد POST ينشئ مورداً
204نجاح بلا محتوىبعد DELETE
400طلب خاطئبيانات لم تمر التحقق
401غير مُصادَقلا رمز أو رمز منتهٍ
403ممنوعمُصادَق لكن بلا صلاحية
404غير موجودالمورد غير موجود
500خطأ خادماستثناء غير متوقع
الفرق بين 401 و403 يُخطئ فيه كثيرون: 401 تعني «لا أعرف من أنت» — سجّل الدخول. 403 تعني «أعرف من أنت، لكنك لا تملك الصلاحية» — لا فائدة من إعادة تسجيل الدخول.

8. جرّبه الآن

npm run dev

# 1. أنشئ حساباً
curl -X POST localhost:3000/api/auth/register \
  -H "Content-Type: application/json" \
  -d '{"email":"a@b.com","password":"secret123"}'

# 2. سجّل الدخول واحصل على الرمز
curl -X POST localhost:3000/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"a@b.com","password":"secret123"}'

# 3. استخدم الرمز
curl localhost:3000/api/tasks \
  -H "Authorization: Bearer ضع_الرمز_هنا"

# 4. اختبر رفض البيانات السيئة
curl -X POST localhost:3000/api/tasks \
  -H "Authorization: Bearer الرمز" \
  -H "Content-Type: application/json" \
  -d '{"title":"ا"}'   # أقصر من 3 أحرف → 400

الاختبار الأخير هو الأهم: تأكد أن API يرفض الخطأ برسالة واضحة، لا أن يعمل فقط عندما تكون المدخلات مثالية. هذا الفرق بين كود تجريبي وكود إنتاج.

9. الخطوات التالية المنطقية

  • قاعدة بيانات حقيقية — PostgreSQL مع prisma أو drizzle بدل المصفوفة.
  • اختبارات — vitest + supertest لتغطية المسارات الحرجة قبل كل نشر.
  • حد للطلبات — express-rate-limit على مسار تسجيل الدخول تحديداً.
  • توثيق تلقائي — ملف OpenAPI من مخططات Zod التي كتبتها بالفعل.
الخطوة التالية: أضف مسار PATCH /api/tasks/:id يحقّق شرطين: أن المهمة موجودة، وأنها تخصّ المستخدم الحالي (req.user.sub). هذا التمرين يعلّمك فحص الصلاحيات — أكثر ثغرة شيوعاً في واجهات API.
