JavaScript للمبتدئين: المتغيرات، الدوال، و DOM

من أول سطر إلى تطبيق قائمة مهام يعمل فعلاً — بلا مكتبات ولا أدوات بناء. جافاسكربت خام، مفهومة سطراً بسطر.

أفريق MyTechol
· 2025-11-24 · 14 دقيقة قراءة · مبتدئ
JS

في الدرس السابق بنيت صفحة ثابتة. المشكلة أنها لا تفعل شيئاً. جافاسكربت هي اللغة التي تجعل الصفحة تتفاعل: تحسب، تتحقق من المدخلات، تحفظ البيانات، وتغيّر ما تراه على الشاشة.

سنتعلمها كما تُستخدم فعلاً: لا نشرح كل خصائص اللغة، بل ما ستكتبه في أول ستة أشهر من عملك.

1. كيف تشغّل الكود

طريقتان، الأولى أسرع للتعلم والثانية للاستخدام الحقيقي:

# الطريقة 1: وحدة تحكم المتصفح
اضغط F12 → تبويب Console → اكتب مباشرة:
2 + 2

# الطريقة 2: ملف خارجي (الطريقة الصحيحة)
<script src="app.js"></script>   ← قبل إغلاق body مباشرة

ضع وسم script في نهاية body لا في head. السبب: الكود ينفّذ فور تحميله، فإذا كان في head فلن يجد عناصر الصفحة بعد.

2. المتغيرات: تخزين قيمة

const name = "عاصم";     // لا تتغير — استخدمها افتراضياً
let age = 25;            // قابلة للتغيير
let isStudent = true;

age = 26;                // مسموح
// name = "سارة";        // خطأ: const لا يعاد تعيينها

قاعدتك العملية: const أولاً. إذا احتجت تغيير القيمة، حوّلها إلى let. لا تستخدم var أبداً في الكود الجديد — سلوكها في النطاق مصدر أخطاء صامتة.

الأنواع الخمسة التي ستستخدمها يومياً:

النوعمثاللاحظ
نص string"مرحباً"بين علامتي تنصيص
رقم number42 / 3.14لا فرق بين صحيح وعشري
منطقي booleantrue / falseبلا علامات تنصيص
مصفوفة array[1, 2, 3]قائمة مرتّبة
كائن object{ name: "علي" }قيم بأسماء

3. الشروط والحلقات

الشرط يجعل البرنامج يتخذ قراراً. استخدم === لا ==: الأول يقارن النوع أيضاً ويمنع مفاجآت غريبة مثل "5" == 5 التي تعطي true.

const score = 78;

if (score >= 90) {
  console.log("ممتاز");
} else if (score >= 60) {
  console.log("ناجح");
} else {
  console.log("يحتاج مراجعة");
}

// شرط مختصر (ternary) — لحالة واحدة بسيطة
const label = score >= 60 ? "ناجح" : "راسب";

والحلقات لتمرير على قائمة:

const langs = ["JavaScript", "Python", "Go"];

// الطريقة الحديثة والمفضّلة
for (const lang of langs) {
  console.log(lang);
}

// مع الفهرس
langs.forEach((lang, i) => {
  console.log(i + 1, lang);
});

4. الدوال: أهم فكرة في البرمجة

الدالة تأخذ مدخلات، تعمل، وتعيد نتيجة. سبب وجودها واحد: كتابة المنطق مرة واحدة واستخدامه مئة مرة.

// تعريف دالة
function greet(name) {
  return "مرحباً " + name;
}

greet("سارة");   // "مرحباً سارة"

// صيغة السهم (الأنيقة والأكثر شيوعاً اليوم)
const greet2 = (name) => `مرحباً ${name}`;

// قيمة افتراضية عند عدم تمرير الوسيط
const greet3 = (name = "زائر") => `مرحباً ${name}`;
greet3();        // "مرحباً زائر"
الفرق العملي: return تُخرج القيمة ليستخدمها الكود بعدها. console.log تطبعها على الشاشة فقط ولا تُعيد شيئاً. الخلط بينهما سبب شائع لظهور undefined.

5. المصفوفات: دوال لا تكتب بلا داعٍ

معظم الكود اليومي هو تحويل قائمة إلى أخرى. هذه الدوال تغنيك عن الحلقات اليدوية:

const products = [
  { name: "لابتوب", price: 4500 },
  { name: "ماوس",   price: 120 },
  { name: "شاشة",   price: 900 }
];

// تحويل كل عنصر
const names = products.map(p => p.name);

// تصفية حسب شرط
const cheap = products.filter(p => p.price < 1000);

// إيجاد عنصر واحد
const mouse = products.find(p => p.name === "ماوس");

// مجموع الأسعار
const total = products.reduce((sum, p) => sum + p.price, 0);

// هل يوجد عنصر يحقق الشرط؟
const hasScreen = products.some(p => p.name === "شاشة");

6. DOM: التحدث مع الصفحة

DOM هو تمثيل المتصفح لصفحتك في الذاكرة. جافاسكربت تقرأ منه وتكتب فيه، فيتغير ما تراه فوراً.

<h1 id="title">عنوان</h1>
<input id="input">
<button id="add">أضف</button>
<ul id="list"></ul>
// 1. اختيار عنصر
const title = document.querySelector("#title");

// 2. قراءة أو تغيير النص
title.textContent = "عنوان جديد";

// 3. تغيير التنسيق
title.style.color = "#00d4ff";

// 4. إضافة كلاس
title.classList.add("highlight");

// 5. قراءة قيمة حقل إدخال
const value = document.querySelector("#input").value;

// 6. الاستماع لحدث النقر
document.querySelector("#add").addEventListener("click", () => {
  console.log("تم النقر");
});

ابحث عن عنصر باستخدام id عند وجوده، لا class. الـ id فريد ويسرّع الاختيار ويمنع الأخطاء.

7. المشروع: قائمة مهام كاملة

هذا تطبيق يعمل بلا مكتبات. انسخه في ملفاتك وشغّله:

<div class="todo">
  <form id="form">
    <input id="input" placeholder="مهمة جديدة..." required>
    <button>أضف</button>
  </form>
  <ul id="list"></ul>
  <p id="count"></p>
</div>

<script>
const form   = document.querySelector("#form");
const input  = document.querySelector("#input");
const list   = document.querySelector("#list");
const count  = document.querySelector("#count");

let tasks = [];           // ذاكرة التطبيق
let nextId = 1;

function render() {
  list.innerHTML = "";

  tasks.forEach(task => {
    const li = document.createElement("li");
    li.textContent = task.title;
    if (task.done) li.style.textDecoration = "line-through";

    // تبديل الحالة
    li.addEventListener("click", () => {
      task.done = !task.done;
      render();
    });

    // حذف
    const del = document.createElement("button");
    del.textContent = "✕";
    del.addEventListener("click", (e) => {
      e.stopPropagation();              // لا تفعّل التبديل
      tasks = tasks.filter(t => t.id !== task.id);
      render();
    });

    li.appendChild(del);
    list.appendChild(li);
  });

  const done = tasks.filter(t => t.done).length;
  count.textContent = `${tasks.length} مهمة · ${done} مكتملة`;
}

form.addEventListener("submit", (e) => {
  e.preventDefault();                    // منع إعادة تحميل الصفحة

  const title = input.value.trim();
  if (!title) return;                    // تجاهل الإدخال الفارغ

  tasks.push({ id: nextId++, title, done: false });
  input.value = "";
  render();
});

render();
</script>

لماذا هذا الترتيب؟ لاحظ ثلاثة قرارات تصنع الفرق بين كود يعمل وكود قابل للصيانة:

  • مصفوفة واحدة كمصدر للحقيقة — الواجهة دائماً نتيجة لها، لا مخزّن مستقل. يمنع اختلاف الشاشة عن البيانات.
  • دالة render واحدة — أي تغيير في البيانات يمر عبرها. لا تحديث يدوي متفرق.
  • البيانات تحمل id — الحذف يعتمد على الهوية لا على الفهرس، فلا تُحذف المهمة الخطأ بعد فرز أو حذف آخر.

8. أخطاء المبتدئين المتكررة

❌ استخدام == بدل ===
❌ نسيان return في الدالة ثم التساؤل عن undefined
❌ تعديل مصفوفة أثناء forEach عليها
❌ قراءة .value قبل وجود العنصر (script في head)
❌ نسيان e.preventDefault() في النموذج فتعيد الصفحة التحميل

عند أي خطأ، افتح Console أولاً. أول سطر أحمر فيه يخبرك بالملف ورقم السطر عادة. اقرأه بدلاً من تخمين الحل.

الخطوة التالية: أضف إلى التطبيق ميزة «حفظ في المتصفح» باستخدام localStorage، بحيث تبقى المهام بعد إعادة تحميل الصفحة. جرّبها بنفسك قبل قراءة الحل — هذا التمرين هو ما يحوّل المعرفة إلى مهارة.
